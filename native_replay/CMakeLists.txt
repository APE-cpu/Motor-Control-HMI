cmake_minimum_required(VERSION 3.20)
project(motor_replay_cpp VERSION 1.0 LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(motor_replay_cpp bindings.cpp ../native_core/src/online_rls.cpp)
target_include_directories(motor_replay_cpp PRIVATE ../native_core/include)
target_compile_features(motor_replay_cpp PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(motor_replay_cpp PRIVATE /W4 /permissive- /EHsc)
endif()
install(TARGETS motor_replay_cpp LIBRARY DESTINATION . RUNTIME DESTINATION .)
